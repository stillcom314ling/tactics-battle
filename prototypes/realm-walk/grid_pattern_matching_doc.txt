Grid Pattern Matching System (C Implementation Plan)

Overview

This system scans a 2D tile grid and detects arbitrary patterns defined
by geometry and constraints.

Supports: - Multiple patterns - Overlapping matches - Real-time
performance (100x100 grid) - Tag-based and relational matching

------------------------------------------------------------------------

Tile

typedef uint32_t TagMask;

typedef struct { int id; TagMask tags; } Tile;

Example tags:

#define TAG_RED (1 << 0) #define TAG_BLUE (1 << 1) #define TAG_GREEN (1
<< 2) #define TAG_BOMB (1 << 3)

Condition Types

Pattern Cell

typedef struct { int dx, dy; ConditionType cond; TagMask tag_mask; int
ref_index; } PatternCell;

------------------------------------------------------------------------

Pattern

typedef struct { int cell_count; PatternCell* cells; int anchor_index;

    int min_dx, max_dx;
    int min_dy, max_dy;

} Pattern;

Matching Function

int match_pattern(Board* b, Pattern* p, int x, int y) { Tile* base_tile
= &b->tiles[y * b->width + x]; Tile* matched[16];

    for (int i = 0; i < p->cell_count; i++) {
        PatternCell* c = &p->cells[i];

        int tx = x + c->dx;
        int ty = y + c->dy;

        if (tx < 0 || ty < 0 || tx >= b->width || ty >= b->height)
            return 0;

        Tile* t = &b->tiles[ty * b->width + tx];
        matched[i] = t;

        switch (c->cond) {
            case COND_ANY:
                break;

            case COND_HAS_TAG:
                if (!(t->tags & c->tag_mask)) return 0;
                break;

            case COND_NOT_TAG:
                if (t->tags & c->tag_mask) return 0;
                break;

            case COND_SAME_AS_ANCHOR:
                if (t->id != base_tile->id) return 0;
                break;

            case COND_SAME_AS_CELL:
                if (t->id != matched[c->ref_index]->id) return 0;
                break;
        }
    }

    return 1;

}
